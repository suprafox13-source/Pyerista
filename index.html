Kod

void setup() {
  pinMode(13, OUTPUT);
}
void loop() {
  digitalWrite(13, HIGH);
  delay(1000);
  digitalWrite(13, LOW);
  delay(1000);
}

🚦 Arduino Trafik Lambası

Kod

int k = 8;
int s = 9;
int y = 10;

void setup() {
  pinMode(k, OUTPUT);
  pinMode(s, OUTPUT);
  pinMode(y, OUTPUT);
}

void loop() {
  digitalWrite(k, HIGH); delay(3000); digitalWrite(k, LOW);
  digitalWrite(s, HIGH); delay(1000); digitalWrite(s, LOW);
  digitalWrite(y, HIGH); delay(3000); digitalWrite(y, LOW);
}
